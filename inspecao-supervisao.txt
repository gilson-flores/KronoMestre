interface StudentSupervisionModalProps {
  isOpen: boolean;
  student: StudentUser | null;
  activities: Activity[];
  onClose: () => void;
  onSaveNotes: (studentId: string, notes: string) => void;
  onSendMessage?: () => void;
  onSimulateStudent?: (student: StudentUser) => void;
  onMarkPhysicalDelivery?: (activityId: string, student: StudentUser, score: number, feedback: string) => void;
  onToggleRecoveryAuthorization?: (recoveryActivityId: string, student: StudentUser) => void;
}

export function formatShortActivityName(act: Activity, allList: Activity[]): string {
  const isRec = Boolean(act.isRecovery || act.type === 'Recuperação Paralela' || act.recoveryFromId);
  if (isRec) {
    const recList = allList.filter(
      (a) => a.isRecovery || a.type === 'Recuperação Paralela' || Boolean(a.recoveryFromId)
    );
    const idx = recList.findIndex((a) => a.id === act.id);
    return `RP${idx >= 0 ? idx + 1 : 1}`;
  }

  const regList = allList.filter(
    (a) => !a.isRecovery && a.type !== 'Recuperação Paralela' && !a.recoveryFromId
  );
  const idx = regList.findIndex((a) => a.id === act.id);
  const num = idx >= 0 ? idx + 1 : 1;
  const modeLabel =
    act.isGameActivity || act.gameData || act.type === 'Jogo Avaliativo'
      ? 'KronoGame'
      : act.type || 'Avaliação';

  return `${modeLabel} AV${num}`;
}

export const StudentSupervisionModal: React.FC<StudentSupervisionModalProps> = ({
  isOpen,
  student,
  activities,
  onClose,
  onSaveNotes,
  onSendMessage,
  onSimulateStudent,
  onMarkPhysicalDelivery,
  onToggleRecoveryAuthorization,
}) => {
  const [notes, setNotes] = useState('');
  const [savedSuccess, setSavedSuccess] = useState(false);
  
  // Estados para lançamento manual dentro de Detalhes
  const [expandedActivityId, setExpandedActivityId] = useState<string | null>(null);
  const [physicalGrade, setPhysicalGrade] = useState<string>('');
  const [physicalFeedback, setPhysicalFeedback] = useState<string>('');

  const studentActivities = useMemo(() => {
    if (!student) return [];
    const isPne = isStudentPNE(student);

    return activities.filter((act) => {
      if (!canStudentAccessActivity(act.turma, student.turma)) return false;
      const adapted = isActivityAdapted(act);
      if (isPne) return adapted || act.includedPneStudentIds?.includes(student.id);
      return !adapted;
    });
  }, [activities, student]);

  useEffect(() => {
    if (student) {
      setNotes(student.pedagogicalNotes || '');
      setSavedSuccess(false);
      setExpandedActivityId(null);
    }
  }, [student]);

  const performanceStats = useMemo(() => {
    if (!student) {
      return { total: 0, gradedCount: 0, pendingCorrectionCount: 0, missingCount: 0, returnedCount: 0, averageGrade: 0, deliveryRate: 0 };
    }

    let gradedCount = 0; let pendingCorrectionCount = 0; let missingCount = 0; let returnedCount = 0; let sumGrades = 0;

    studentActivities.forEach((act) => {
      const sub = act.submissions?.find(
        (s) => s.studentId === student.id || (Boolean(student.matricula) && s.matricula === student.matricula) || s.studentName?.toLowerCase() === student.fullName?.toLowerCase()

      if (!sub || sub.status === 'draft') missingCount++;
      else if (sub.status === 'graded') {
        gradedCount++;
        sumGrades += (Number(sub.objectiveScore) || 0) + (Number(sub.essayScore) || 0) + (Number(sub.extraPoints) || 0);
      }
      else if (sub.status === 'pending') pendingCorrectionCount++;
      else if (sub.status === 'returned') returnedCount++;
    });

    const total = studentActivities.length;
    const delivered = gradedCount + pendingCorrectionCount;
    return {
      total, gradedCount, pendingCorrectionCount, missingCount, returnedCount,
      averageGrade: gradedCount > 0 ? Number((sumGrades / gradedCount).toFixed(1)) : 0,
      deliveryRate: total > 0 ? Math.round((delivered / total) * 100) : 0,

    onSaveNotes(student.id, notes);
    setSavedSuccess(true);
    setTimeout(() => setSavedSuccess(false), 3000);

    if (!onMarkPhysicalDelivery) return;
    const parsedGrade = parseFloat(physicalGrade.replace(',', '.'));
    
    if (isNaN(parsedGrade) || parsedGrade < 0 || parsedGrade > 10) {
      alert('Por favor, insira uma nota válida entre 0 e 10.');
      return;

    setPhysicalGrade('');
    setPhysicalFeedback('');
    setExpandedActivityId(null); // Recolhe após lançar
  };

  const toggleExpand = (actId: string) => {
    if (expandedActivityId === actId) {
      setExpandedActivityId(null);
    } else {
      setExpandedActivityId(actId);
      setPhysicalGrade('');
      setPhysicalFeedback('');
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/85 backdrop-blur-md flex items-center justify-center p-2.5 sm:p-4 animate-fadeIn">
      <div className="bg-surface border border-border rounded-3xl w-full max-w-4xl max-h-[94vh] flex flex-col shadow-2xl overflow-hidden">
        
        {/* TOP HEADER */}
        <div className="p-4 sm:p-6 border-b border-border bg-background/70 flex flex-col sm:flex-row sm:items-center justify-between gap-4 shrink-0">
          <div className="flex items-center gap-3.5 min-w-0">
            <img src={student.avatarUrl || 'https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?w=100'} alt={student.fullName} className="w-12 h-12 sm:w-14 sm:h-14 rounded-full object-cover border-2 border-primary shrink-0 shadow-md" />
            <div className="min-w-0">
              <div className="flex items-center gap-1.5 flex-wrap">

                <span className="text-[11px] font-bold px-2 py-0.5 rounded-md bg-surface border border-border text-foreground">Turma {student.turma}</span>
                {student.tag && <span className="text-[10px] font-bold px-2 py-0.5 rounded-md bg-amber-500/20 text-amber-300 border border-amber-500/30">{student.tag}</span>}

        <div className="p-4 sm:p-6 overflow-y-auto custom-scrollbar flex-1 space-y-6">

              <div className="text-xl sm:text-2xl font-black text-foreground">{performanceStats.gradedCount > 0 ? performanceStats.averageGrade.toFixed(1) : '-'}<span className="text-xs font-normal text-muted"> / 10</span></div>

                return (
                  <div key={act.id} className="p-3.5 rounded-2xl bg-background border border-border space-y-2.5">
                    <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2.5">
                      <div className="flex items-center gap-2 flex-wrap min-w-0">
                        <span className={`text-xs font-black px-2.5 py-1 rounded-lg border ${isRecovery ? 'bg-amber-500/20 text-amber-300 border-amber-500/40' : 'bg-primary/20 text-primary border-primary/40'}`}>{shortName}</span>
                        
                        {sub?.isPhysicalDelivery && <span className="text-[10px] font-bold px-2 py-0.5 rounded-md bg-cyan-500/20 text-cyan-300 border border-cyan-500/40">📄 Analógica</span>}

'bg-emerald-500/20 text-emerald-300 border-emerald-500/40 hover:bg-rose-500/20 hover:text-rose-300' : 'bg-amber-500/20 text-amber-300 border-amber-500/40 hover:bg-amber-500/30'}`}>

                        )}
                        <button type="button" onClick={() => toggleExpand(act.id)} className="px-3 py-1.5 rounded-xl bg-surface hover:bg-surface-hover text-muted hover:text-foreground border border-border text-[11px] font-bold flex items-center gap-1 cursor-pointer">
                          <span>Detalhes</span>{isExpanded ? <ChevronUp className="w-3.5 h-3.5" /> : <ChevronDown className="w-3.5 h-3.5" />}

                          <p className="text-muted">Prazo: {act.deadline || 'Sem data'}</p>
                        </div>
                        
                        {/* Se já foi corrigida, exibe as notas */}
                        {sub && sub.status === 'graded' && (
                          <div className="pt-3 border-t border-border space-y-1.5">
                            <p className="text-emerald-400 font-bold">Atividade Concluída e Avaliada.</p>
                            <p className="text-muted">Objetiva: <strong>{sub.objectiveScore?.toFixed(1) || 0}</strong> | Dissertativa: <strong>{sub.essayScore?.toFixed(1) || 0}</strong></p>
                            {sub.essayAnswer && <p className="text-foreground italic bg-background p-2 rounded-lg border border-border">&ldquo;{sub.essayAnswer}&rdquo;</p>}
                            {sub.teacherFeedback && <p className="text-primary mt-2"><strong>Parecer:</strong> {sub.teacherFeedback}</p>}
                          </div>
                        )}

                        {/* Se estiver pendente ou devolvida, mostra formulário manual */}
                        {(!sub || sub.status === 'returned' || sub.status === 'pending') && (
                          <div className="pt-3 border-t border-border space-y-3">
                            <div className="flex items-center gap-2">
                              <FileCheck className="w-4 h-4 text-cyan-400 shrink-0" />
                              <h4 className="font-bold text-cyan-400">Lançamento de Atividade Analógica / Feita Manualmente</h4>

                              </div>

                              <div className="sm:col-span-3">
                                <button type="button" onClick={() => handleConfirmPhysicalSelect(act.id)} className="w-full h-[36px] rounded-xl bg-cyan-500 hover:bg-cyan-400 text-slate-950 font-black text-xs flex items-center justify-center gap-1.5 transition-all cursor-pointer shadow-md">
                                  <CheckCircle2 className="w-4 h-4" /><span>Lançar</span>

                          </div>

                      </div>

                  </div>
                );
              })}
            </div>
          </div>

          <div className="space-y-2 pt-2 border-t border-border">
            <label className="text-xs font-bold text-muted uppercase tracking-wider block">Anotações Pedagógicas do Aluno:</label>
            <textarea rows={3} value={notes} onChange={(e) => setNotes(e.target.value)} placeholder="Registre observações individuais ou acompanhamentos..." className="w-full bg-background border border-border rounded-2xl p-3 text-xs text-foreground focus:outline-none focus:border-primary" />
            <div className="flex items-center justify-between">
              {savedSuccess ? <span className="text-xs font-bold text-emerald-400 flex items-center gap-1"><CheckCircle2 className="w-4 h-4" /> Salvo com sucesso!</span> : <span />}
              <button type="button" onClick={handleSave} className="px-5 py-2.5 rounded-xl bg-primary hover:bg-primary-hover text-white font-bold text-xs flex items-center gap-1.5 cursor-pointer shadow-md"><Save className="w-4 h-4" /><span>Salvar Anotações</span></button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
